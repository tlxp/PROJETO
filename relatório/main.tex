  \begin{center}%

% Texto introdutório + diagrama na mesma página, sem float (evita espaço vazio).
\newcommand{\introdiagram}[5][\relax]{%
  \Needspace{\dimexpr 0.75\textheight + 6\baselineskip\relax}%
  \begin{samepage}%
  #2\par\vspace{0.2em}\nopagebreak[4]%
  \diagram@render[#1]{#3}{#4}{#5}%
  \end{samepage}%
}
% Subsecção + texto + diagrama colados (evita subtítulo órfão no fim da página).
\newcommand{\diagramsubsection}[6][\relax]{%
  \Needspace{\dimexpr 0.75\textheight + 10\baselineskip\relax}%
  \begin{samepage}%
  \begingroup
  \setbeforesubsecskip{1.2ex plus 0.3ex minus .1ex}%
  \setaftersubsecskip{0.35ex plus 0.1ex minus .05ex}%
  \subsection{#2}%
  \endgroup
  \nopagebreak[4]%
  #3\par\vspace{0.15em}\nopagebreak[4]%
  \diagram@render[#1]{#4}{#5}{#6}%
  \end{samepage}%
}
\makeatother
